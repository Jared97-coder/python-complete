# ================================================================
# COMANDOS ÚTILES
# ================================================================
#
# Ejecutar todos los tests:
#   pytest
#
# Con verbose:
#   pytest -v
#
# Con verbose + stdout:
#   pytest -v -s
#
# Solo tests rápidos (sin "slow"):
#   pytest -m "not slow"
#
# Solo tests de integración:
#   pytest -m integration
#
# Con cobertura:
#   pytest --cov=src --cov-report=html
#   # Abrir: htmlcov/index.html
#
# En paralelo (4 CPUs):
#   pytest -n 4
#
# Auto (usar todos CPUs):
#   pytest -n auto
#
# Detener en primer fallo:
#   pytest -x
#
# Último fallo primero:
#   pytest --lf
#
# Solo fallos:
#   pytest --failed-first
#
# Con timeout (10s por test):
#   pytest --timeout=10
#
# Watch mode (re-run al cambiar):
#   ptw
