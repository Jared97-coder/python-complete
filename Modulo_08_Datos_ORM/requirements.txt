# ----------------------------------------------------------------------------
# Notas de Instalación
# ----------------------------------------------------------------------------

# Windows con SQL Server:
#   1. Instalar ODBC Driver 17 for SQL Server
#   2. pip install pyodbc

# PostgreSQL en producción:
#   - Usar psycopg2 (no psycopg2-binary)
#   - Compilar desde source: pip install psycopg2

# MongoDB:
#   - Requiere MongoDB server corriendo
#   - Docker: docker run -d -p 27017:27017 mongo
#   - Instalación: https://www.mongodb.com/try/download/community

# MySQL/MariaDB:
#   - pip install pymysql (pure Python)
#   - pip install mysqlclient (requiere compilación)

# ----------------------------------------------------------------------------
# Connection Strings Ejemplo
# ----------------------------------------------------------------------------

# SQLite:
#   sqlite:///database.db          # Archivo local
#   sqlite:///:memory:             # En memoria

# PostgreSQL:
#   postgresql://user:pass@localhost:5432/mydb
#   postgresql+psycopg2://user:pass@localhost/mydb
#   postgresql+asyncpg://user:pass@localhost/mydb  # Async

# SQL Server:
#   mssql+pyodbc://user:pass@host/db?driver=ODBC+Driver+17+for+SQL+Server
#   mssql+pymssql://user:pass@host/db

# MySQL:
#   mysql+pymysql://user:pass@localhost/mydb
#   mysql+mysqlclient://user:pass@localhost/mydb

# MongoDB:
#   mongodb://localhost:27017/
#   mongodb://user:pass@host:27017/mydb
